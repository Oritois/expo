import React, { useState } from 'react';
import { Text, View, Image, TouchableOpacity, StyleSheet, ScrollView } from 'react-native';

export default function App() {
  const [selectedWord, setSelectedWord] = useState(null);

  const data = {
    "نیوتن": {
      image: "https://upload.wikimedia.org/wikipedia/commons/d/d1/Sir_Isaac_Newton.jpg",
      description: "نیوتن در سال ۱۶۸۷ قانون گرانش را مطرح کرد."
    },
    "انیشتین": {
      image: "https://upload.wikimedia.org/wikipedia/commons/d/d3/Albert_Einstein_Head.jpg",
      description: "انیشتین نظریه نسبیت را معرفی کرد."
    }
  };

  return (
    <ScrollView style={styles.container}>
      <Text style={styles.title}>روی یکی از واژه‌ها بزن:</Text>

      <View style={styles.wordsContainer}>
        {Object.keys(data).map((word, index) => (
          <TouchableOpacity key={index} onPress={() => setSelectedWord(word)}>
            <Text style={styles.word}>{word}</Text>
          </TouchableOpacity>
        ))}
      </View>

      {selectedWord && (
        <View style={styles.result}>
          <Image source={{ uri: data[selectedWord].image }} style={styles.image} />
          <Text style={styles.description}>{data[selectedWord].description}</Text>
        </View>
      )}
    </ScrollView>
  );
}

const styles = StyleSheet.create({
  container: {
    marginTop: 50,
    padding: 20,
    backgroundColor: '#121212',
  },
  title: {
    color: '#fff',
    fontSize: 22,
    marginBottom: 20,
  },
  wordsContainer: {
    flexDirection: 'row',
    flexWrap: 'wrap',
    marginBottom: 30,
  },
  word: {
    backgroundColor: '#333',
    color: '#fff',
    padding: 10,
    borderRadius: 10,
    margin: 5,
  },
  result: {
    alignItems: 'center',
  },
  image: {
    width: 250,
    height: 250,
    borderRadius: 20,
    marginBottom: 10,
  },
  description: {
    color: '#ccc',
    fontSize: 18,
    textAlign: 'center',
  },
});
